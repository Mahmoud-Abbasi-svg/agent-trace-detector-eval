def lab(r): return int(r["category"] == "attacked")
def benign_vocab(recs): return {s["tool"] for r in recs if r["category"] == "benign" for s in r["steps"]}


def coverage(V, recs):
    occ = [s["tool"] in V for r in recs for s in r["steps"]]
    zero = [not any(s["tool"] in V for s in r["steps"]) for r in recs]
    return {"n_traces": len(recs), "occ_coverage": round(float(np.mean(occ)), 4),
            "frac_traces_zero_known": round(float(np.mean(zero)), 4)}


def by_class(V, recs):
    out = {"all": coverage(V, recs)}
    for c, name in [(1, "attacked"), (0, "not_attacked")]:
        sub = [r for r in recs if lab(r) == c]
        if sub: out[name] = coverage(V, sub)
    return out
